# ------------------------------------------------------------
# 数据库配置
# ------------------------------------------------------------
# MySQL/TiDB 数据库连接字符串
# 格式: mysql://用户名:密码@主机:端口/数据库名?ssl={"rejectUnauthorized":true}
DATABASE_URL=mysql://root:password@localhost:3306/avatar_companion

# ------------------------------------------------------------
# 认证与安全
# ------------------------------------------------------------
# JWT 会话 Cookie 签名密钥（建议使用随机生成的长字符串）
JWT_SECRET=your-jwt-secret-key-here

# Manus OAuth 应用 ID
VITE_APP_ID=your-manus-app-id

# Manus OAuth 后端服务地址
OAUTH_SERVER_URL=https://api.manus.im

# Manus OAuth 登录门户地址（前端使用）
VITE_OAUTH_PORTAL_URL=https://id.manus.im

# 项目所有者 Manus OpenID（用于管理员权限自动分配）
OWNER_OPEN_ID=your-owner-open-id

# 项目所有者名称
OWNER_NAME=your-owner-name

# ------------------------------------------------------------
# Manus 内置 API 服务（LLM、存储、语音转录、图片生成、通知等）
# ------------------------------------------------------------
# 内置 API 服务地址（服务端使用）
BUILT_IN_FORGE_API_URL=https://forge.manus.im

# 内置 API 服务密钥（服务端使用，Bearer Token）
BUILT_IN_FORGE_API_KEY=your-forge-api-key

# 内置 API 服务地址（前端使用）
VITE_FRONTEND_FORGE_API_URL=https://forge.manus.im

# 内置 API 服务密钥（前端使用，仅限公开安全的操作）
VITE_FRONTEND_FORGE_API_KEY=your-frontend-forge-api-key

# ------------------------------------------------------------
# 分析与监控（可选）
# ------------------------------------------------------------
# Umami 分析服务端点
VITE_ANALYTICS_ENDPOINT=

# Umami 网站 ID
VITE_ANALYTICS_WEBSITE_ID=

# ------------------------------------------------------------
# 应用外观配置（可选）
# ------------------------------------------------------------
# 应用标题
VITE_APP_TITLE=3D Avatar Studio

# 应用 Logo URL
VITE_APP_LOGO=

# ------------------------------------------------------------
# 运行时配置（可选）
# ------------------------------------------------------------
# 运行环境：development | production
NODE_ENV=development

# 服务端口号（默认 3000）
PORT=3000
